% ---------- COLUMN 1: setup + question ----------
\begin{column}{\onecolwid}
\begin{minipage}[t][\colheight][s]{\linewidth}
\Large

\begin{block}{Motivation}
  \begin{itemize}
    \item Deep learning solves \emphcolor{high-dimensional} dynamic models once thought intractable.
    \item The networks are \emphcolor{massively overparameterized}: far more parameters than ``data'' points.
    \item Classical worry: \emphcolor{``overfitting''} out of grid.
    \item Difference: residuals are deterministic, so \emphcolor{no noise to overfit}. What matters is \emphcolor{between the grid points}.
  \end{itemize}
\end{block}

\vfill

\begin{block}{The question}
  When we solve dynamic models with neural networks, is overparameterization a \emphcolor{necessary evil}, or does it buy us something?
\end{block}

\vfill

\begin{block}{Setup: models as functional equations}
  An economic model is an equation (Euler or Bellman):
  \vskip0.5ex
  \begin{eqbox}
  $\displaystyle \ell(x,f^\star)=\mathbf{0}$
  \end{eqbox}
  \vskip0.5ex
  Fit $f(\cdot;\theta)$ by minimizing the residual loss
  \vskip0.5ex
  \begin{eqbox}
  $\displaystyle \mathcal{L}(\theta;\mathcal{D}) = \frac{1}{N}\sum_{x\in\mathcal{D}} \big\| \ell\big(x,f(\cdot;\theta)\big)\big\|_2^2$
  \end{eqbox}
  \vskip0.5ex
  Neural nets: \emphcolor{overparameterized} ($M\gg N$), \emphcolor{non-convex}. Solution depends on \emphcolor{initialization}.
\end{block}

\vfill

\begin{block}{What we find}
  {\textcolor{Sienna}{\textbf{Empirically}}}
  \begin{itemize}
    \item \textbf{Accuracy improves with size.}
    \item \textbf{Stability improves with size:} the seed spread collapses.
  \end{itemize}
  \vskip0.5ex
  {\textcolor{Sienna}{\textbf{Theoretically}}}
  \begin{itemize}
    \item \textbf{Width buys agreement; the implicit bias decides what you agree on.}
  \end{itemize}
\end{block}

\vfill

\begin{block}{Four models of increasing difficulty}
  \begin{itemize}
    \item \emphcolor{Linear-Quadratic:} closed form, sharpest test.
    \item \emphcolor{McCall:} 1D, kink at the reservation wage.
    \item \emphcolor{RBC:} 2D stochastic, no closed form (vs.\ VFI).
    \item \emphcolor{Small Open Economy:} 5D, beyond any grid.
  \end{itemize}
  \vskip0.5ex
  Here: the first two, plus the random-features dissection.
\end{block}

\end{minipage}
\end{column}

% ---------- COLUMN 2: basic results ----------
\begin{column}{\onecolwid}
\begin{minipage}[t][\colheight][s]{\linewidth}
\large

\begin{block}{The two worked models}
  \emphcolor{Linear-Quadratic:} firm chooses investment $u$, quadratic adjustment cost, state $(y,Y)$.
  \vskip0.5ex
  \begin{eqbox}
  $\begin{array}{c}
  \displaystyle v(y,Y)=\max_{u}\ (\alpha_0-\alpha_1 Y)y-\tfrac{\gamma}{2}u^2+\beta\,v(y',Y')\\[1.2ex]
  \text{s.t.}\quad y'=y+u,\qquad Y'=h_0+h_1 Y
  \end{array}$
  \end{eqbox}
  \vskip1ex
  \emphcolor{McCall:} worker accepts offer $w\sim f$ on $[0,B]$ or keeps searching.
  \vskip0.5ex
  \begin{eqbox}
  $\displaystyle v(w)=\max\!\left\{\tfrac{w}{1-\beta},\ c+\beta\!\int_0^B v(w')f(w')\,dw'\right\}$
  \end{eqbox}
\end{block}

\vfill

\begin{block}{Result 1 -- Linear-Quadratic regulator}
  \emphcolor{Closed form}, so error is exact. Euler residual on $n_Y=6$ points; train and test loss \emphcolor{indistinguishable} at every width.
  \vskip1ex
  \centering
  \includegraphics[width=0.95\textwidth]{LQ_policy_rel_error_median_vs_max_tanh}
  \vskip1ex
  \raggedright
  Median and max error fall to $\sim10^{-4}$. The \emphcolor{seed band narrows}.
\end{block}

\vfill

\begin{block}{Result 2 -- McCall job search}
  \emphcolor{Non-smooth:} a \emphcolor{kink} at the reservation wage, learned from only $N=11$ points.
  \vskip1ex
  \centering
  \includegraphics[width=0.95\textwidth]{McCall_value_function_24_vs_128_tanh}
  \vskip1ex
  \raggedright
  Narrow ($H=24$): seeds disagree near the kink. Wide ($H=128$): all 50 \emphcolor{converge to the same function}. \emphcolor{Stability confirmed.}
\end{block}

\end{minipage}
\end{column}

% ---------- COLUMN 3: careful examination ----------
\begin{column}{\onecolwid}
\begin{minipage}[t][\colheight][s]{\linewidth}

\begin{block}{A careful examination: random features}
  \emphcolor{Freeze} the first layer, train only the outer weights $v$:
  \vskip0.5ex
  \begin{eqbox}
  $\displaystyle f_H(Y;\theta)=\sum_{j} v_j\,\sigma(\tilde{w}_j Y + \tilde{b}_j) + c$
  \end{eqbox}
  \vskip0.5ex
  The LQ residual is \emphcolor{linear in $\theta$}: $\Phi\theta=d$ with $H+1>N$, so \emphcolor{infinitely many solutions}. \emphcolor{The algorithm is the identifying assumption}. Gradient descent from zero picks
  \vskip0.5ex
  \begin{eqbox}
  $\displaystyle \theta^\star = \operatorname*{argmin}_{\theta:\,\Phi\theta=d}\ \|\theta\|_2 = \Phi^\top(\Phi\Phi^\top)^{-1}d,$
  \end{eqbox}
  \vskip0.5ex
  computed directly, \emphcolor{bypassing training}.
\end{block}

\vfill

\begin{block}{Minimum norm knows no economic model}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}\centering
      \includegraphics[width=\textwidth]{LQ_policy_RF_lstsq_N6_tanh_test_train_loss}\\[0.2ex]
      {\footnotesize \emphcolor{Tanh works:} test loss stays flat.}
    \end{column}
    \begin{column}{0.5\textwidth}\centering
      \includegraphics[width=\textwidth]{LQ_policy_RF_lstsq_N6_relu_test_train_loss}\\[0.2ex]
      {\footnotesize \emphcolor{ReLU fails:} test loss rises with width.}
    \end{column}
  \end{columns}
  \vskip1ex
  Train loss is zero for both; min-norm in \emphcolor{coefficients}, not in \emphcolor{functions}:
  \begin{itemize}
    \item \emphcolor{Tanh:} smooth features, the fit stays linear.
    \item \emphcolor{ReLU:} kinks inject curvature between points.
  \end{itemize}
\end{block}

\vfill

\begin{block}{The cure: penalize the kinks between points}
  \centering
  \includegraphics[width=\curw]{relu_bumps_illustration}\\[0.2ex]
  {\footnotesize $u(x)-u^\star(x)$: the fit kinks between grid points.}
  \vskip1.5ex
  \includegraphics[width=\curw]{LQ_policy_RF_lstsq_penalty_N6_relu_rel_error_median_vs_max}\\[0.2ex]
  {\footnotesize Regularized: flat at $10^{-15}$ at every width.}
  \vskip1ex
  \raggedright
  Penalize the interior kinks (curvature between points): the fit \emphcolor{snaps back} to the truth.
\end{block}

\vfill

\begin{block}{Conclusion}
  \begin{itemize}
    \item Robust across \emphcolor{four models} and smooth activations.
    \item \emphcolor{The parameter count was never the risk.}
  \end{itemize}
\end{block}
